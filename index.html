// Authentication Actions (Updated with Loading & Clear Error Handling) $('loginBtn').onclick = async () => { const email = $('email').value.trim(); const password = $('password').value; if (!email || password.length < 6) { showStatus('authMsg', 'Kripya sahi email aur kam se kam 6 aksharon ka password darj karein.'); return; } $('loginBtn').disabled = true; showStatus('authMsg', 'Login ho raha hai, kripya pratiksha karein...', true); try { if (!sb) { sb = window.supabase.createClient(SUPABASE_URL, SUPABASE_ANON_KEY); } const { data, error } = await sb.auth.signInWithPassword({ email, password }); if (error) { if (error.message.includes("Invalid login credentials")) { showStatus('authMsg', 'Galat email/password ya account nahi mila! Pehle "नया खाता बनाएँ" par click karke account banayein.'); } else { showStatus('authMsg', 'Login Error: ' + error.message); } } else { showStatus('authMsg', 'Login safal!', true); } } catch (err) { showStatus('authMsg', 'Error: ' + (err.message || err)); } finally { $('loginBtn').disabled = false; } }; $('signupBtn').onclick = async () => { const email = $('email').value.trim(); const password = $('password').value; if (!email || password.length < 6) { showStatus('authMsg', 'Kripya sahi email aur kam se kam 6 aksharon ka password darj karein.'); return; } $('signupBtn').disabled = true; showStatus('authMsg', 'Account banaya ja raha hai...', true); try { if (!sb) { sb = window.supabase.createClient(SUPABASE_URL, SUPABASE_ANON_KEY); } const { data, error } = await sb.auth.signUp({ email, password }); if (error) { showStatus('authMsg', 'Signup Error: ' + error.message); } else if (data.user && !data.session) { showStatus('authMsg', 'Account ban gaya! Supabase me Email Confirmation on hai to apne Gmail par jaakar verify link click karein, fir login karein.', true); } else { showStatus('authMsg', 'Account safaltapoorvak ban gaya aur login ho gaya!', true); } } catch (err) { showStatus('authMsg', 'Error: ' + (err.message || err)); } finally { $('signupBtn').disabled = false; } };
